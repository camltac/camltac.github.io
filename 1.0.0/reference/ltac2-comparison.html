Comparison with Ltac2#

The tables below map Ltac2 constructs to their equivalent in Camltac.

Types#

Type declarations#

Ltac2

OCaml

 
Ltac2 Type t := int.

type t = int

Ltac2 Type rec nat := [O | S (nat)].

type nat = O | S of nat

Ltac2 Type flag := {
  mutable used : bool
}.

type flag = { mutable used: bool }

Ltac2 Type rec tree :=
  [Node (nat, forest)]
with forest :=
  [Leaf (nat) | Cons (tree, forest)].

type tree = Node of nat * forest
and forest =
| Leaf of nat
| Cons of tree * forest

Ltac2 Type exn ::=
  [ My_error (string) ].

exception My_error of string

Ltac2 Import Type nat as n.

type n = nat

Ltac2

| S ?n => n
| _ => Control.zero Not_found
end.

  (Not_found, Exninfo.null)

| {|S ?n|} -> return n

  (Not_found, Exninfo.null)

lazy_match! goal with
| [ H : _ |- _ ] => apply H
end.

match%lazy goal with
| { h = _ }, _ ->
  let* env in
  let h = Ltac2.Control.hyp env h in
  apply [term h]

match! reverse goal with
| [ H : _ |- _ ] => apply H
end.

match%rocq reverse goal with
| { h = _ }, _ ->
  let* env in

  apply [term h]
